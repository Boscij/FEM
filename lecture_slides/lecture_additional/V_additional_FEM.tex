\begin{frame}{Dimensionen aus den drei Grundbeziehungen}
  \footnotesize
  In jeder Beziehung muss das Matrix-Vektor-Produkt aufgehen.
  Daraus folgen die Dimensionen von $\Nmat$, $\Bmat$ und $\ke$.

  \vspace{0.3em}
  \begin{tcolorbox}[colback=blue!6, colframe=blue!70!black, boxrule=0.8pt,
      left=6pt, right=6pt, top=2pt, bottom=2pt]
    \textbf{1.\ Verschiebungsfeld} $\widetilde{\bm{u}} = \Nmat\,\bm{v}^e$:
    \;\;
    $\underbrace{\widetilde{\bm{u}}}_{\hlo{\text{Versch.-Komp.}}}
      = \Nmat \cdot \underbrace{\bm{v}^e}_{\hlb{\text{Element-FG}}}$
    \;\;$\Longrightarrow$\;\;
    $\Nmat:\;\hlo{\text{Versch.-Komp.}} \times \hlb{\text{Element-FG}}$
  \end{tcolorbox}

  \begin{tcolorbox}[colback=green!6, colframe=green!55!black, boxrule=0.8pt,
      left=6pt, right=6pt, top=2pt, bottom=2pt]
    \textbf{2.\ Verzerrungsfeld} $\bm{\varepsilon} = \Bmat\,\bm{v}^e$:
    \;\;
    $\underbrace{\bm{\varepsilon}}_{\hlg{\text{Dehnungs-Komp.}}}
      = \Bmat \cdot \underbrace{\bm{v}^e}_{\hlb{\text{Element-FG}}}$
    \;\;$\Longrightarrow$\;\;
    $\Bmat:\;\hlg{\text{Dehnungs-Komp.}} \times \hlb{\text{Element-FG}}$
  \end{tcolorbox}

  \begin{tcolorbox}[colback=orange!6, colframe=orange!80!black, boxrule=0.8pt,
      left=6pt, right=6pt, top=2pt, bottom=2pt]
    \textbf{3.\ Element-Steifigkeit} $\ke = \int_{\Omega^e}\Bmat^{\!\top}\Emat\,\Bmat\,\dd\Omega$:
    \;\;
    $\underbrace{\Bmat^{\!\top}}_{\hlb{\text{FG}}\times\hlg{\text{Deh}}}
      \underbrace{\Emat}_{\hlg{\text{Deh}\times\text{Deh}}}
      \underbrace{\Bmat}_{\hlg{\text{Deh}}\times\hlb{\text{FG}}}$
    \;\;$\Longrightarrow$\;\;
    $\ke:\;\hlb{\text{Element-FG}}\times\hlb{\text{Element-FG}}$
    (quadratisch)
  \end{tcolorbox}

  \vspace{0.2em}
  \centering
  \scriptsize
  \textbf{Element-FG} = Anzahl Knoten-Verschiebungen je Element = Länge von $\bm{v}^e$
\end{frame}
